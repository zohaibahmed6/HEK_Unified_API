					@pPostalPostCodeID=NULL,

			-1, --[InvoiceDetailID]
			0, --[InvoiceTransactionID]
			@vMasterServiceSubServiceID, --[MasterServiceSubServiceID] Y
			@vAppointmentServiceID, --[AppointmentServiceID] Y [Lookup].[uspLoadPOCtransaction]

			@IsFiledNote=0,
			@pIsAutoPay=0,
			@pIsReportInvoice=0,

			@outputParam=@vInvoiceNo OUTPUT

			DECLARE @vDummyInvoiceNo INT = -1

			exec [PoC].[UspTransactionInvoicePaymentInsertUpdateNew] 
			@pInvoiceTransactionID=0,
			@pInvoiceID=@vInvoiceNo,
			@pDate=@vCurrentDate,
			@pPatientID=@pPatientId,

	BEGIN CATCH

		BEGIN
